async function identity(path) {
  const core = await readFile(resolve(path, 'build/ring64.wasm'));
  const allHost = createHash('sha256'), productionHost = createHash('sha256'), hostFiles = [], benchmarkFiles = [];
  for (const name of (await readdir(resolve(path, 'build/host'))).filter((name) => name.endsWith('.js')).sort()) {
    const bytes = await readFile(resolve(path, 'build/host', name));
    allHost.update(name + '\0'); allHost.update(bytes);
    if (!name.endsWith('.test.js')) { productionHost.update(name + '\0'); productionHost.update(bytes); }
    hostFiles.push({ name, bytes: bytes.length, sha256: hash(bytes) });
  }
  for (const name of (await readdir(resolve(path, 'benchmarks'))).filter((name) => name.endsWith('.mjs')).sort()) {
    const bytes = await readFile(resolve(path, 'benchmarks', name)); benchmarkFiles.push({ name, bytes: bytes.length, sha256: hash(bytes) });
  }
  const metadata = await readFile(resolve(path, 'paging-source.json'));
  return { path, source: JSON.parse(metadata).source, metadataSha256: hash(metadata), coreSha256: hash(core), coreBytes: core.length, hostSha256: allHost.digest('hex'), productionHostSha256: productionHost.digest('hex'), hostFiles, benchmarkFiles };
}

function inspectPlacement(instance, fields) {
  const ex = instance.exports; check(typeof ex.core_state_ptr === 'function' && ex.memory instanceof WebAssembly.Memory, 'observer did not capture Core exports');
  const ptr = ex.core_state_ptr() >>> 0;
  const descriptor = JSON.parse(new TextDecoder().decode(new Uint8Array(ex.memory.buffer, ex.core_info_ptr() >>> 0, ex.core_info_len() >>> 0)));
  equal({ offset: descriptor.x87?.offset, registers: descriptor.x87?.registers, registerBytes: descriptor.x87?.registerBytes }, { offset: 456, registers: 0, registerBytes: 10 }, 'authoritative x87 layout');
  const bank = ptr + descriptor.x87.offset, registers = bank + descriptor.x87.registers;
  check(Number.isSafeInteger(ptr) && ptr + descriptor.size <= ex.memory.buffer.byteLength, 'state pointer outside memory');
  return { ...fields, statePointer: ptr, statePointerHex: '0x' + ptr.toString(16), statePointerMod16: ptr % 16, statePointerMod64: ptr % 64, stateBytes: descriptor.size, x87: { offset: descriptor.x87.offset, bytes: descriptor.x87.bytes, registerBytes: descriptor.x87.registerBytes, bankBase: bank, bankBaseMod16: bank % 16, bankBaseMod64: bank % 64, initialTop: (new DataView(ex.memory.buffer).getUint16(bank + descriptor.x87.statusWord, true) >>> 11) & 7, physicalSlots: [3, 4].map((slot) => ({ slot, ...byteRange(registers + slot * descriptor.x87.registerBytes, descriptor.x87.registerBytes) })) } };
}
