const uleb = (input) => {
  check(Number.isSafeInteger(input) && input >= 0 && input <= 0xffffffff, 'ULEB input outside u32');
  const bytes = []; let value = input;
  do { const low = value % 128; value = Math.floor(value / 128); bytes.push(low | (value ? 128 : 0)); } while (value);
  return Uint8Array.from(bytes);
};
function readUleb(bytes, start) {
  let value = 0, scale = 1;
  for (let i = 0; i < 5; i++, scale *= 128) {
    const byte = bytes[start + i]; check(byte !== undefined, 'truncated ULEB'); value += (byte & 127) * scale;
    if (!(byte & 128)) { check(value <= 0xffffffff, 'ULEB overflow'); return { value, next: start + i + 1 }; }
  }
  throw Error('overlong ULEB');
}
function readString(bytes, start) {
  const n = readUleb(bytes, start), end = n.next + n.value;
  check(end <= bytes.length, 'truncated string');
  return { value: new TextDecoder().decode(bytes.subarray(n.next, end)), next: end };
}
function sections(bytes) {
  check(hex(bytes.subarray(0, 8)) === '0061736d01000000', 'unexpected module header');
  const found = [], seen = new Set(); let at = 8;
  while (at < bytes.length) {
    const start = at, id = bytes[at++], n = readUleb(bytes, at), end = n.next + n.value;
    check(end <= bytes.length, 'truncated section');
    if (id) { check(!seen.has(id), 'duplicate section'); seen.add(id); }
    found.push({ id, start, payload: n.next, end }); at = end;
  }
  return found;
}
function patternOffsets(bytes, needle) {
  const offsets = [];
  for (let at = 0; at <= bytes.length - needle.length; at++) {
    if (needle.every((value, index) => bytes[at + index] === value)) offsets.push(at);
  }
  return offsets;
}
function unique(bytes, needle, name) {
  const offsets = patternOffsets(bytes, needle);
  check(offsets.length === 1, `${name}: expected unique exact pattern, got ${offsets.length}`);
  return offsets[0];
}

// Only the generated run function is admitted. The full Core is never rewritten.
function parseUnit(bytes) {
  check(WebAssembly.validate(bytes), 'original generated unit is invalid');
  const ss = sections(bytes), find = (id) => { const s = ss.find((s) => s.id === id); check(s, `missing section${id}`); return bytes.subarray(s.payload, s.end); };
  const typeBytes = find(1), types = []; let p = readUleb(typeBytes, 0); const typeCount = p.value; let at = p.next;
  for (let i = 0; i < typeCount; i++) {
    check(typeBytes[at++] === 0x60, 'only plain function types expected');
    p = readUleb(typeBytes, at); at = p.next; const params = Array.from(typeBytes.subarray(at, at + p.value)); at += p.value;
    p = readUleb(typeBytes, at); at = p.next; const returns = Array.from(typeBytes.subarray(at, at + p.value)); at += p.value;
    types.push({ params, returns });
  }
  check(at === typeBytes.length, 'type section tail');
  const f = find(3), fnCount = readUleb(f, 0); check(fnCount.value === 1, 'expected exactly one defined run function');
  const fnType = readUleb(f, fnCount.next); check(fnType.next === f.length, 'function section tail');
  const signature = types[fnType.value];
  check(JSON.stringify(signature?.params) === JSON.stringify([0x7f, 0x7e, 0x7e, 0x7f, 0x7f]) && signature.returns.length === 1 && signature.returns[0] === 0x7f, 'unexpected five-argument run ABI');
  const imports = find(2), importNames = []; p = readUleb(imports, 0); at = p.next;
  for (let i = 0; i < p.value; i++) {
    let item = readString(imports, at); at = item.next; const module = item.value;
    item = readString(imports, at); at = item.next; const name = item.value, kind = imports[at++];
    importNames.push(`${module}.${name}`);
    if (kind === 0) at = readUleb(imports, at).next;
    else if (kind === 2) { const flags = readUleb(imports, at); at = flags.next; check(!(flags.value & ~1), 'unexpected memory import flags'); at = readUleb(imports, at).next; if (flags.value & 1) at = readUleb(imports, at).next; }
    else throw Error('unexpected generated import kind');
  }
  check(at === imports.length, 'import section tail');
  check(!importNames.some((s) => /\.(processor|paging_control)$/.test(s)), 'counterfactual scope excludes timestamp and MSR helpers');
  const code = ss.find((s) => s.id === 10); check(code, 'missing CodeSection');
  const payload = bytes.subarray(code.payload, code.end), count = readUleb(payload, 0); check(count.value === 1, 'expected one code body');
  const n = readUleb(payload, count.next), body = payload.subarray(n.next, n.next + n.value); check(n.next + n.value === payload.length, 'code body tail');
  p = readUleb(body, 0); at = p.next; const locals = signature.params.slice();
  for (let i = 0; i < p.value; i++) { const group = readUleb(body, at); at = group.next; const type = body[at++]; check(group.value < 10000, 'excessive local group'); locals.push(...Array(group.value).fill(type)); }
  check(locals[R] === 0x7e && locals[REASON] === 0x7f, 'live R/reason local types do not match the validated source contract');
  return { ss, code, body, declarationsEnd: at, imports: importNames, signature, localCount: locals.length };
}
function patchUnit(bytes, mode) {
  const unit = parseUnit(bytes), original = unit.body, code = original.subarray(unit.declarationsEnd);
  const where = unique(code, epilogue, '18-byte clock epilogue'), load = unique(code, entry, 'entry B-to-R load');
  check(hex(code.subarray(-3)) === hex(suffix), 'final LocalGet REASON25/End suffix changed');
  check(load + entry.length < where, 'entry load must precede common publication');
  check(JSON.stringify(patternOffsets(code, Uint8Array.from(get(B)))) === JSON.stringify([load, where + 8]), 'B must be used only by the entry load and original epilogue');
  let changed = code;
  if (mode === 'omit') changed = concat(code.subarray(0, where), code.subarray(where + epilogue.length));
  else if (mode === 'move-after-publication') {
    const removed = concat(code.subarray(0, where), code.subarray(where + epilogue.length));
    changed = concat(removed.subarray(0, -3), epilogue, removed.subarray(-3));
  } else if (mode === 'future-bias') {
    changed = concat(code.subarray(0, load + entry.length), futureEntry, code.subarray(load + entry.length, where), futureExit, code.subarray(where + epilogue.length));
    check(patternOffsets(changed, epilogue).length === 0, 'old epilogue survived future rebase');
    check(JSON.stringify(patternOffsets(changed, Uint8Array.from(get(B)))) === JSON.stringify([load, load + entry.length + 8]), 'future B use escaped the entry credit');
  } else check(mode === 'as-is', 'unknown patch mode');
  const body = concat(original.subarray(0, unit.declarationsEnd), changed);
  const payload = concat(uleb(1), uleb(body.length), body);
  const output = concat(bytes.subarray(0, unit.code.start), [10], uleb(payload.length), payload, bytes.subarray(unit.code.end));
  check(WebAssembly.validate(output), `${mode}: patched generated module is invalid`);
  const oldNonCode = unit.ss.filter((s) => s.id !== 10).map((s) => hash(bytes.subarray(s.start, s.end)));
  const newNonCode = sections(output).filter((s) => s.id !== 10).map((s) => hash(output.subarray(s.start, s.end)));
  check(JSON.stringify(oldNonCode) === JSON.stringify(newNonCode), 'non-Code sections changed');
  if (mode === 'as-is') check(hash(output) === hash(bytes), 'symmetric as-is wrapper changed bytes');
  return { output, proof: { mode, originalSha256: hash(bytes), compiledSha256: hash(output), originalBytes: bytes.length, compiledBytes: output.length, originalBodyBytes: original.length, compiledBodyBytes: body.length, epilogueInstructionOffset: where, entryInstructionOffset: load, nonCodeSectionsUnchanged: true, originalValidated: true, compiledValidated: true, imports: unit.imports } };
}
